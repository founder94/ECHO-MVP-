# PROPOSAL ONLY (not under .github/workflows, not active, no live workflow changed).
# Install as .github/workflows/queue-adapter.yml only after review + 대표 approval of the items listed at the bottom.
#
# What it does (all code is in tools/automation, run from main, never from PR head):
#   init   (workflow_dispatch mode=init, inputs.tasks = JSON [{id,ref:"pr:N",owner:{login:"founder94",type:"User"}}])
#            -> CLI `init`: sender must be the real founder94 User; creates the approved READY queue once (never overwrites).
#   start  (workflow_dispatch mode=start, sender founder94 User) -> START for the first approved READY task (worker identity = github-actions[bot], separate field).
#   events (pull_request_review submitted, issue_comment created) -> CLI verifies repo + CURRENT PR head (gh api, right before) +
#            actor + marker/native findings, then CAS-saves state and outbox entries in one commit.
#   drain  -> CLI `drain` dispatches outbox entries (START/FIX/REVIEW) one at a time, claim-by-CAS, result recorded DONE/FAILED.
# State lives only on the non-protected branch refs/heads/echo-automation-state (plain fast-forward push, no force; stale writer is rejected).
name: queue-adapter
on:
  workflow_dispatch:
    inputs:
      mode: { description: 'init | start', required: true, type: choice, options: [init, start] }
      tasks: { description: 'init only: JSON array of approved tasks', required: false, type: string }
  pull_request_review: { types: [submitted] }
  issue_comment: { types: [created] }
permissions:
  contents: write        # ONLY to push refs/heads/echo-automation-state (needs approval)
  pull-requests: write   # `gh pr comment` for dispatch + `gh api pulls/..` reads (needs approval)
  issues: read
concurrency:
  group: queue-adapter   # serializes runs; the store's fast-forward push is the actual lock
  cancel-in-progress: false
jobs:
  adapter:
    runs-on: ubuntu-latest
    timeout-minutes: 10      # cost cap: no model call, no paid API, only node + gh
    steps:
      - uses: actions/checkout@v4
        with: { ref: main, persist-credentials: true }   # reviewed code from main only
      - uses: actions/setup-node@v4
        with: { node-version: 22 }
      - name: handle event
        id: handle
        env:
          EVENT_NAME: ${{ github.event_name }}
          MODE: ${{ github.event.inputs.mode }}
          DELIVERY: ${{ github.run_id }}-${{ github.run_attempt }}
          GITHUB_REPOSITORY: ${{ github.repository }}
          GH_TOKEN: ${{ github.token }}              # existing token only; never printed
          QUEUE_STORE_DIR: ${{ github.workspace }}
          QUEUE_MAX_ROUNDS: '5'                      # approved cap, controller refuses anything above 5
          QUEUE_WORKER_TRIGGER: ${{ vars.QUEUE_WORKER_TRIGGER }}   # approved trigger text (repository variable, not a secret); empty => dispatch fails closed
        run: |
          set +e
          if [ "$EVENT_NAME" = "workflow_dispatch" ] && [ "$MODE" = "init" ]; then
            node tools/automation/queue-adapter-cli.cjs init "$GITHUB_EVENT_PATH" "$DELIVERY"; code=$?
          else
            node tools/automation/queue-adapter-cli.cjs "$EVENT_NAME" "$GITHUB_EVENT_PATH" "$DELIVERY"; code=$?
          fi
          echo "code=$code" >> "$GITHUB_OUTPUT"
          # exit 2 = queue halted (STOP): recorded, not a crash. exit 1 = fail closed: job fails, nothing dispatched.
          [ "$code" = "0" ] || [ "$code" = "2" ]
      - name: drain outbox (dispatch START/FIX/REVIEW)
        if: steps.handle.outputs.code == '0' && github.event.inputs.mode != 'init'
        env:
          GITHUB_REPOSITORY: ${{ github.repository }}
          GH_TOKEN: ${{ github.token }}
          QUEUE_STORE_DIR: ${{ github.workspace }}
          QUEUE_WORKER_TRIGGER: ${{ vars.QUEUE_WORKER_TRIGGER }}
        run: node tools/automation/queue-adapter-cli.cjs drain   # exit 3 = dispatch failed/uncertain: job fails visibly
# ---------------------------------------------------------------------------------------------------------------------

#  refuses to continue (needs a human); dispatch failure is stored FAILED (max 3 attempts), never DONE. REVIEW is not posted by default
